### ۳. اسکریپت اجرایی لینوکس با کامنت‌های خط‌به‌خط (`codes.sh`)

```bash
#!/bin/bash
# ==============================================================================
# اسکریپت تمرینی دستورات پایه Git برای اجرا در ترمینال
# ==============================================================================

# ۱. ساخت یک پوشه تمرینی جدید
mkdir git_practice

# ۲. ورود به پوشه ساخته شده (cd = Change Directory)
cd git_practice

# ۳. راه‌اندازی گیت در این پوشه (ساخت مخزن محلی)
git init

# ۴. ساخت یک فایل متنی نمونه با دستور echo
echo "Gene Name: BRCA1" > gene_info.txt

# ۵. مشاهده وضعیت فایل‌های پوشه (فایل gene_info.txt به رنگ قرمز نشان داده می‌شود)
git status

# ۶. انتقال فایل به مرحله آماده‌سازی (Staging)
git add gene_info.txt

# ۷. ثبت نهایی تغییرات همراه با پیام توضیحی
git commit -m "افزودن اطلاعات ژن BRCA1"

# ۸. دیدن تاریخچه تمام ثبت‌های انجام شده (Commit History)
git log --oneline
